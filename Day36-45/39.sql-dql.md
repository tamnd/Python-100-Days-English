## DQL in Detail

Next, we will use the school course-selection database created earlier to explain query operations in DQL. Querying is extremely important for both developers and data analysts, because it determines whether we can retrieve the data we actually need from a relational database. It is recommended that you rerun the DDL from the earlier lesson that created the database and tables, as well as the DML from the previous lesson that inserted data, so that the tables and data are correct before executing the operations below.

```sql
USE school;

-- Query all information for all students
SELECT stu_id,
       stu_name,
       stu_sex,
       stu_birth,
       stu_addr,
       col_id
  FROM tb_student;

-- Query student ID, name, and hometown
SELECT stu_id AS StudentID,
       stu_name AS Name,
       stu_addr AS Hometown
  FROM tb_student;

-- Query course names and credits
SELECT cou_name AS CourseName,
       cou_credit AS Credits
  FROM tb_course;

-- Query all female students' names and birth dates
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0;

-- Query female students from Chengdu, Sichuan
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       AND stu_addr = 'Chengdu, Sichuan';

-- Query students who are female or from Chengdu, Sichuan
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       OR stu_addr = 'Chengdu, Sichuan';

-- Query all students born in the 1980s
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE stu_birth BETWEEN '1980-1-1' AND '1989-12-31';

-- Query courses with more than 2 credits
SELECT cou_name,
       cou_credit
  FROM tb_course
 WHERE cou_credit > 2;

-- Query courses whose credits are odd numbers
SELECT cou_name,
       cou_credit
  FROM tb_course
 WHERE cou_credit MOD 2 <> 0;

-- Query student IDs for students who chose course 1111 and scored above 90
SELECT stu_id
  FROM tb_record
 WHERE cou_id = 1111
       AND score > 90;

-- Query the name and sex of the student named Yang Guo
SELECT stu_name AS Name,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name = 'Yang Guo';

SELECT stu_name AS Name,
       IF(stu_sex, 'Male', 'Female') AS Gender
  FROM tb_student
 WHERE stu_name = 'Yang Guo';

-- Query students whose family name is Yang
SELECT stu_name AS Name,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang%';

-- Query students named Yang whose given names are one character long
SELECT stu_name AS Name,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang_';

-- Query students named Yang whose given names are two characters long
SELECT stu_name AS Name,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang__';

-- Query students whose student IDs end with 3
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_id LIKE '%3';

-- Query students whose names contain Bu or Yan
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Bu%'
       OR stu_name LIKE '%Yan%';

SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Bu%'
 UNION
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Yan%';

-- Query students whose family name is Yang or Lin and whose names have three Chinese characters total
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name REGEXP '[LinYang][\\u4e00-\\u9fa5]{2}';

-- Query students whose hometown was not entered
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) = ''
       OR stu_addr is null;

-- Query students whose hometown was entered
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Query all course-selection dates
SELECT DISTINCT sel_date
  FROM tb_record;

-- Query hometowns without duplicates
SELECT DISTINCT stu_addr
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Query male students' names and birth dates ordered from oldest to youngest
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY stu_birth ASC;

-- Convert birthdays to ages
SELECT stu_name AS Name,
       FLOOR(DATEDIFF(CURDATE(), stu_birth) / 365) AS Age
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY Age DESC;

-- Query the birth date of the oldest student
SELECT MIN(stu_birth)
  FROM tb_student;

-- Query the birth date of the youngest student
SELECT MAX(stu_birth)
  FROM tb_student;

-- Query the highest score in course 1111
SELECT MAX(score)
  FROM tb_record
 WHERE cou_id = 1111;

-- Query minimum score, maximum score, average score, standard deviation, and variance for student 1001
SELECT MIN(score) AS MinScore,
       MAX(score) AS MaxScore,
       ROUND(AVG(score), 1) AS AvgScore,
       STDDEV(score) AS StdDev,
       VARIANCE(score) AS Variance
  FROM tb_record
 WHERE stu_id = 1001;

-- Query the average score of student 1001, treating null as 0
SELECT ROUND(SUM(score) / COUNT(*), 1) AS AvgScore
  FROM tb_record
 WHERE stu_id = 1001;

-- Query the number of male and female students
SELECT CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender,
       COUNT(*) AS Total
  FROM tb_student
 GROUP BY stu_sex;

-- Query the number of students in each college
SELECT col_id AS CollegeID,
       COUNT(*) AS Total
  FROM tb_student
 GROUP BY col_id
  WITH ROLLUP;

-- Query the number of male and female students in each college
SELECT col_id AS CollegeID,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender,
       COUNT(*) AS Total
  FROM tb_student
 GROUP BY col_id, stu_sex;

-- Query each student's student ID and average score
SELECT stu_id AS StudentID,
       ROUND(AVG(score), 1) AS AvgScore
  FROM tb_record
 GROUP BY stu_id;

-- Query students whose average score is at least 90
SELECT stu_id AS StudentID,
       ROUND(AVG(score), 1) AS AvgScore
  FROM tb_record
 GROUP BY stu_id
HAVING AvgScore >= 90;

-- Query students whose average score in courses 1111, 2222, and 3333 is at least 90
SELECT stu_id AS StudentID,
       ROUND(AVG(score), 1) AS AvgScore
  FROM tb_record
 WHERE cou_id in (1111, 2222, 3333)
 GROUP BY stu_id
HAVING AvgScore >= 90
 ORDER BY AvgScore ASC;

-- Query the name of the oldest student
SELECT stu_name
  FROM tb_student
 WHERE stu_birth = (SELECT MIN(stu_birth)
                      FROM tb_student);

-- Query the names of students who selected more than two courses
SELECT stu_name
  FROM tb_student
 WHERE stu_id in (SELECT stu_id
                    FROM tb_record
                   GROUP BY stu_id
                  HAVING COUNT(*) > 2);

-- Query student names, birth dates, and the names of their colleges
SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student AS t1, tb_college AS t2
 WHERE t1.col_id = t2.col_id;

SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student INNER JOIN tb_college
       ON tb_student.col_id = tb_college.col_id;

SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student NATURAL JOIN tb_college;

SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student CROSS JOIN tb_college;

-- Query student names, course names, and scores
SELECT stu_name,
       cou_name,
       score
  FROM tb_student, tb_course, tb_record
 WHERE tb_student.stu_id = tb_record.stu_id
       AND tb_course.cou_id = tb_record.cou_id
       AND score is not null;

SELECT stu_name,
       cou_name,
       score
  FROM tb_student
       INNER JOIN tb_record
           ON tb_student.stu_id = tb_record.stu_id
       INNER JOIN tb_course
           ON tb_course.cou_id = tb_record.cou_id
 WHERE score is not null;

SELECT stu_name,
       cou_name,
       score
  FROM tb_student
       NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null;

-- Supplement: take the first 5 rows
SELECT stu_name,
       cou_name,
       score
  FROM tb_student
       NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5;

-- Supplement: take rows 6 through 10
SELECT stu_name,
       cou_name,
       score
  FROM tb_student
       NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5
OFFSET 5;

-- Supplement: take rows 11 through 15
SELECT stu_name,
       cou_name,
       score
  FROM tb_student
       NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 10, 5;

-- Query student names and average scores
SELECT stu_name,
       avg_score
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            ROUND(AVG(score), 1) AS avg_score
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Query student names and the number of selected courses
SELECT stu_name,
       total
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            COUNT(*) AS total
                       FROM tb_record
                      GROUP BY stu_id) as tmp;
```

### Summary

DQL is used to query data. Through selection, projection, sorting, grouping, aggregation, subqueries, joins, and pagination, we can retrieve exactly the information we need from relational data.
